    private static Color c(int rgb, int alpha) {
        return new Color((rgb >> 16) & 255, (rgb >> 8) & 255, rgb & 255, alpha);
    }

    private static Color mix(Color a, Color b, float t) {
        return new Color(Math.round(a.getRed() + (b.getRed() - a.getRed()) * t), Math.round(a.getGreen() + (b.getGreen() - a.getGreen()) * t), Math.round(a.getBlue() + (b.getBlue() - a.getBlue()) * t));
    }

    private static Graphics2D open(BufferedImage image) {
        Graphics2D g = image.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g.setRenderingHint(RenderingHints.KEY_STROKE_CONTROL, RenderingHints.VALUE_STROKE_PURE);
        return g;
    }

    /** Adds grain: every pixel's brightness moves by up to {@code amount}. */
    private static void grain(BufferedImage image, int amount, long seed) {
        Random r = new Random(seed);
        for (int y = 0; y < image.getHeight(); y++) {
            for (int x = 0; x < image.getWidth(); x++) {
                int rgb = image.getRGB(x, y);
                int d = r.nextInt(amount * 2 + 1) - amount;
                int rr = Math.max(0, Math.min(255, ((rgb >> 16) & 255) + d));
                int gg = Math.max(0, Math.min(255, ((rgb >> 8) & 255) + d));
                int bb = Math.max(0, Math.min(255, (rgb & 255) + d));
                image.setRGB(x, y, (rr << 16) | (gg << 8) | bb);
            }
        }
    }

    private static void tracked(Graphics2D g, String text, double cx, double y, float tracking) {
        FontMetrics fm = g.getFontMetrics();
        double total = 0;
        for (char ch : text.toCharArray()) total += fm.charWidth(ch) + tracking;
        total -= tracking;
        double x = cx - total / 2;
        for (char ch : text.toCharArray()) {
            g.drawString(String.valueOf(ch), (float) x, (float) y);
            x += fm.charWidth(ch) + tracking;
        }
    }

    private static void title(Graphics2D g, String title, String subtitle, Color main, Color sub, Color shadow) {
        g.setFont(new Font("Serif", Font.BOLD, 56));
        g.setColor(shadow);
        tracked(g, title, 802, 100, 9);
        g.setColor(main);
        tracked(g, title, 800, 98, 9);
        g.setFont(new Font("SansSerif", Font.PLAIN, 18));
        g.setColor(sub);
        tracked(g, subtitle, 800, 134, 5);
    }

    private static void goldFrame(Graphics2D g, Color a, Color b, Color inner) {
        g.setPaint(new LinearGradientPaint(0, 0, W, H, new float[]{0f, 0.5f, 1f}, new Color[]{a, b, a}));
        g.setStroke(new BasicStroke(6f));
        g.draw(new RoundRectangle2D.Float(22, 22, W - 44, H - 44, 22, 22));
        g.setColor(inner);
        g.setStroke(new BasicStroke(2f));
        g.draw(new RoundRectangle2D.Float(34, 34, W - 68, H - 68, 17, 17));

        g.setPaint(new LinearGradientPaint(0, 0, 30, 30, new float[]{0f, 1f}, new Color[]{a, b}));

    }

    /** The frame of one tile place, drawn as the template draws it: a panel, a dark inset for the picture, an inset for the name and a numbered badge. */
    private static void tileFrame(Graphics2D g, Slot s, Color border, Color panel, Color inset, Color nameInset, Color badge, Color badgeText) {
        int[] b = s.bounds();
        int[] ib = s.icon();
        int[] nb = s.name();
        if (s.round()) return;
        RoundRectangle2D frame = new RoundRectangle2D.Float(b[0], b[1], b[2], b[3], 15, 15);
        g.setColor(new Color(0, 0, 0, 90));
        g.fill(new RoundRectangle2D.Float(b[0] + 2, b[1] + 4, b[2], b[3], 15, 15));
        g.setColor(panel);
        g.fill(frame);
        g.setColor(border);
        g.setStroke(new BasicStroke(3.2f));
        g.draw(frame);
        g.setColor(new Color(255, 255, 255, 40));
        g.setStroke(new BasicStroke(1f));
        g.draw(new RoundRectangle2D.Float(b[0] + 3, b[1] + 3, b[2] - 6, b[3] - 6, 12, 12));
        g.setColor(inset);
        g.fill(new RoundRectangle2D.Float(ib[0] - 3, ib[1] - 3, ib[2] + 6, ib[3] + 6, 8, 8));
        g.setColor(new Color(255, 255, 255, 25));
        g.draw(new RoundRectangle2D.Float(ib[0] - 3, ib[1] - 3, ib[2] + 6, ib[3] + 6, 8, 8));
        g.setColor(nameInset);
        g.fill(new RoundRectangle2D.Float(nb[0] - 1, nb[1] - 1, nb[2] + 2, nb[3] + 2, 6, 6));
        g.setColor(new Color(255, 255, 255, 40));
        g.draw(new RoundRectangle2D.Float(nb[0] - 1, nb[1] - 1, nb[2] + 2, nb[3] + 2, 6, 6));
        badge(g, b[0] + 9, b[1] + 9, s.number(), badge, badgeText);
    }

    private static void badge(Graphics2D g, int cx, int cy, int number, Color fill, Color text) {
        g.setColor(new Color(0, 0, 0, 100));
        g.fill(new Ellipse2D.Float(cx - 12, cy - 11, 25, 25));
        g.setColor(fill);
        g.fill(new Ellipse2D.Float(cx - 13, cy - 13, 26, 26));
        g.setColor(new Color(255, 255, 255, 90));
        g.setStroke(new BasicStroke(1.2f));
        g.draw(new Ellipse2D.Float(cx - 11, cy - 11, 22, 22));
        g.setFont(new Font("SansSerif", Font.BOLD, 12));
        g.setColor(text);
        String n = Integer.toString(number);
        FontMetrics fm = g.getFontMetrics();
        g.drawString(n, cx - fm.stringWidth(n) / 2f, cy + (fm.getAscent() - fm.getDescent()) / 2f);
    }

    private static Slot slot(BingoLayout l, int n) {
        return l.slots().get(n - 1);
    }

    private static double[] centre(Slot s) {
        return new double[]{s.bounds()[0] + s.bounds()[2] / 2.0, s.bounds()[1] + s.bounds()[3] / 2.0};
    }

            double base = 820 + layer * 46;

    private static void leaf(Graphics2D g, double x, double y, double angle, double size, Random r) {
        AffineTransform old = g.getTransform();
        g.translate(x, y);
        g.rotate(angle);
        float tone = r.nextFloat();
        Color body = mix(c(0x1f6a3a, 255), c(0x6fbf5a, 255), tone);
        g.setColor(new Color(body.getRed(), body.getGreen(), body.getBlue(), 120 + r.nextInt(100)));
        Path2D shape = new Path2D.Double();
        shape.moveTo(0, 0);
        shape.curveTo(size * 0.5, -size * 0.6, size * 1.3, -size * 0.3, size * 1.6, 0);
        shape.curveTo(size * 1.3, size * 0.3, size * 0.5, size * 0.6, 0, 0);
        g.fill(shape);
        g.setColor(new Color(0.8f, 1f, 0.8f, 0.25f));
        g.setStroke(new BasicStroke(0.8f));
        g.draw(new Line2D.Double(0, 0, size * 1.5, 0));
        g.setTransform(old);
    }

    /** A 640 by 400 picture: a shape's whole board scaled down, or the top of a taller grid board. */
    private static void writePreview(BufferedImage source, Path file) throws Exception {
        int w = 640;
        int h = (int) Math.round(source.getHeight() * (w / (double) source.getWidth()));
        BufferedImage scaled = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
        Graphics2D g = scaled.createGraphics();
        g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BICUBIC);
        g.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
        g.drawImage(source, 0, 0, w, h, null);
        g.dispose();
        BufferedImage cut = scaled.getSubimage(0, 0, w, Math.min(h, 400));
        var writer = ImageIO.getImageWritersByFormatName("jpg").next();
        var params = writer.getDefaultWriteParam();
        params.setCompressionMode(javax.imageio.ImageWriteParam.MODE_EXPLICIT);
        params.setCompressionQuality(0.8f);
        try (var os = ImageIO.createImageOutputStream(file.toFile())) {
            writer.setOutput(os);
            writer.write(null, new javax.imageio.IIOImage(cut, null, null), params);
        }
        writer.dispose();
    }

            params.setCompressionQuality(0.93f);
            try (var out = ImageIO.createImageOutputStream(resources.resolve(entry.getKey() + ".jpg").toFile())) {
                writer.setOutput(out);
                writer.write(null, new javax.imageio.IIOImage(entry.getValue(), null, null), params);
            }
            writer.dispose();
            List<TileView> tiles = new ArrayList<>();
            for (int i = 0; i < l.tiles(); i++) {
                String[] s = sample[i % sample.length];
                boolean done = i % 4 == 0;
                boolean locked = !done && i % 3 == 1;
                tiles.add(new TileView(i + 1, s[0], 25 + (i % 5) * 25, done, done ? 1 + i % 3 : 0, 6, s[1], done ? (i % 2 == 0 ? "Metorrite" : "Atrum Dies") : null, locked));
            }
            Files.write(OUT.resolve(entry.getKey() + ".png"), BingoBoardImage.render(ImageIO.read(resources.resolve(entry.getKey() + ".jpg").toFile()), l, tiles, false));
        }
    }
